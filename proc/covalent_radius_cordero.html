public pure function covalent_radius_cordero(atomic_number) result(radius)

Cordero covalent radius in Angstrom, or 0 where none is tabulated

Zero is a deliberate refusal rather than a default. A caller doing bond perception must decide what to do about an element with no radius; picking one here would mean guessing bonds for superheavies from a number nobody measured.

   pure function covalent_radius_cordero(atomic_number) result(radius)
      !! Cordero covalent radius in Angstrom, or 0 where none is tabulated
      !!
      !! Zero is a deliberate refusal rather than a default. A caller doing
      !! bond perception must decide what to do about an element with no
      !! radius; picking one here would mean guessing bonds for superheavies
      !! from a number nobody measured.
      integer, intent(in) :: atomic_number
      real(dp) :: radius

      select case (atomic_number)
      case (1:MAX_Z_CORDERO)
         radius = CORDERO_RADII(atomic_number)
      case default
         radius = 0.0_dp
      end select
   end function covalent_radius_cordero
