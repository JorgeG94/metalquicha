   pure function compare(a, b, n) result(order)
      !! Lexicographic order on two zero-padded monomer rows
      integer, intent(in) :: a(:), b(:)
      integer, intent(in) :: n
      integer :: order

      integer :: i

      order = 0
      do i = 1, n
         if (a(i) < b(i)) then
            order = -1
            return
         end if
         if (a(i) > b(i)) then
            order = 1
            return
         end if
      end do
   end function compare
