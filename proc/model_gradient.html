   pure subroutine model_gradient(a, b, c, n, g)
      !! `dE/dc_k = b_k + sum_j c_j (A_kj + A_jk)`
      real(dp), intent(in) :: a(:, :), b(:), c(:)
      integer, intent(in) :: n
      real(dp), intent(out) :: g(:)
      integer :: k, j

      do k = 1, n
         g(k) = b(k)
         do j = 1, n
            g(k) = g(k) + c(j)*(a(k, j) + a(j, k))
         end do
      end do
   end subroutine model_gradient
