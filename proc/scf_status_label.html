   pure function scf_status_label(status) result(text)
      !! One word for a status, for a table column or a log line
      integer, intent(in) :: status
      character(len=3) :: text

      select case (status)
      case (SCF_CONVERGED)
         text = "yes"
      case (SCF_NOT_CONVERGED)
         text = "NO "
      case default
         text = "?  "
      end select
   end function scf_status_label
