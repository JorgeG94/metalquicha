   subroutine hash_text(this, value)
      class(hasher_t), intent(inout) :: this
      character(len=*), intent(in) :: value

      integer :: i

      call this%int(len(value))
      do i = 1, len(value)
         call this%byte(int(iachar(value(i:i)), int64))
      end do
   end subroutine hash_text
